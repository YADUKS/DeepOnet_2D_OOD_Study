# 2. Generate once, then dependent data checks/OOD generation.
# Use one CPU for FE generation: the user reported successful OOD completion
# with this configuration after a multi-threaded job failed with signal 11.
# The precise native crash cause is unconfirmed. Keep crash diagnostics enabled.
# These thread limits apply to the submitted generation jobs only; plotting and
# GPU submissions retain their usual CPU allocations. Cached FE cases are reused.
GEN=$(sbatch --parsable --account="$STUDY_ACCOUNT" --cpus-per-task=1 --export=ALL,PYTHONFAULTHANDLER=1,OPENBLAS_NUM_THREADS=1,MKL_NUM_THREADS=1 jobs/sunbird/cpu.sh generate --config study.json)
OOD=$(sbatch --parsable --account="$STUDY_ACCOUNT" --cpus-per-task=1 --export=ALL,PYTHONFAULTHANDLER=1,OPENBLAS_NUM_THREADS=1,MKL_NUM_THREADS=1 --dependency="afterok:$GEN" jobs/sunbird/cpu.sh generate-ood --config study.json)
sbatch --account="scw7010" --dependency="afterok:$GEN" jobs/sunbird/cpu.sh plot-sampling --config study.json
sbatch --account="scw7010" --dependency="afterok:$GEN" jobs/sunbird/cpu.sh preview --config study.json --family circle --case 0
sbatch --account="scw7010" --dependency="afterok:$GEN" jobs/sunbird/cpu.sh mesh-check --config study.json

# 3. Tanh grid: 25 cells, one cell per GPU task (seeds within a cell run serially).
TANH=$(sbatch --parsable --account="scw7010" --dependency="afterok:$GEN" --array=0-24 --output=outputs/slurm/tanh_%A_%a.out jobs/sunbird/gpu.sh tune --config study.json --stage-name grid)
TANH_SELECT=$(sbatch --parsable --account="scw7010" --dependency="afterok:$TANH" jobs/sunbird/cpu.sh tune-select --config study.json --stage-name grid)

# 3b. Optional detailed Tanh report (one GPU evaluation + CPU-only figures).
# In study.json, report_field_methods=["cartesian","fe_vertices"] enables both
# field plots; ["cartesian"] or ["fe_vertices"] enables just that method.
# The initial FE evaluation reads saved mesh caches and queries existing weights.
# After that, switching methods needs only the CPU figures job. Grid metrics and
# best/median/worst rankings remain unchanged. See docs/manual/tuning_results.md.
# Submit together with the active grid/OOD jobs; no array is needed here.
# If all 25 cells and OOD generation already finished, omit this old-job dependency.
REPORT_EVAL=$(sbatch --parsable --account="$STUDY_ACCOUNT" --dependency="afterok:$TANH:$OOD" jobs/sunbird/gpu.sh report-tuning --config study.json --stage-name grid --report-phase evaluate)
REPORT_FIGURES=$(sbatch --parsable --account="$STUDY_ACCOUNT" --dependency="afterok:$REPORT_EVAL" jobs/sunbird/cpu.sh report-tuning --config study.json --stage-name grid --report-phase figures)

# 4. GELU grid, followed by validation-only selection.
GELU=$(sbatch --parsable --account="scw7010" --dependency="afterok:$TANH_SELECT" --array=0-24 --output=outputs/slurm/gelu_%A_%a.out jobs/sunbird/gpu.sh tune --config study.json --stage-name grid_gelu)
GELU_SELECT=$(sbatch --parsable --account="scw7010" --dependency="afterok:$GELU" jobs/sunbird/cpu.sh tune-select --config study.json --stage-name grid_gelu)

# 5. Three learning rates at the better architecture/activation winner.
LR=$(sbatch --parsable --account="scw7010" --dependency="afterok:$GELU_SELECT" --array=0-2 --output=outputs/slurm/lr_%A_%a.out jobs/sunbird/gpu.sh tune --config study.json --stage-name lr)
LR_SELECT=$(sbatch --parsable --account="scw7010" --dependency="afterok:$LR" jobs/sunbird/cpu.sh tune-select --config study.json --stage-name lr)

# 6. Five sizes x five seeds. Task k uses sizes[k//5], seeds[k%5].
# Change array bounds if changing the number of sizes/seeds/grid cells.
SCALE=$(sbatch --parsable --account="scw7010" --dependency="afterok:$LR_SELECT" --array=0-24 --output=outputs/slurm/scaling_%A_%a.out jobs/sunbird/gpu.sh train-scaling --config study.json)

# 7. Reports after every final run has completed.
sbatch --account="scw7010" --dependency="afterok:$SCALE" jobs/sunbird/cpu.sh scaling-report --config study.json

done
